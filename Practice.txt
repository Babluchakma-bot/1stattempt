marks = float(input("Enter your mark: "))

grades = {
    80: "A+",
    70: "A",
    60: "A-",
    50: "B"
}

for threshold, grade in grades.items():
    if marks >= threshold:
        print(grade)
        break
else:
    print("Below average")
